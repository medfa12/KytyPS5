cmake_minimum_required(VERSION 3.20)
project(vsh_subprocess_tests LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The initial subprocess backend exercises Windows process/job/pipe semantics")
endif()
enable_testing()
add_library(vsh_subprocess_component STATIC
  ../../src/loader/runtimeChildProtocol.cpp
  ../../src/loader/runtimeChildInput.cpp
  ../../src/loader/runtimeChildFrames.cpp
  ../../src/loader/runtimeChildClient.cpp
  ../../src/loader/runtimeChildProcess.cpp
  ../../src/loader/runtimeChildManifest.cpp)
target_sources(vsh_subprocess_component PRIVATE ../../src/graphics/presentation/window/runtimeWindowBridge.cpp)
target_include_directories(vsh_subprocess_component PUBLIC ../../src)
target_compile_features(vsh_subprocess_component PUBLIC cxx_std_20)
target_compile_definitions(vsh_subprocess_component PUBLIC KYTY_SUBPROCESS_TESTS=1 NOMINMAX)
add_executable(vsh_subprocess_probe probe.cpp)
target_link_libraries(vsh_subprocess_probe PRIVATE vsh_subprocess_component onecore)
add_executable(vsh_subprocess_tests process_tests.cpp)
target_compile_definitions(vsh_subprocess_tests PRIVATE KYTY_SUBPROCESS_TESTS=1 NOMINMAX)
target_link_libraries(vsh_subprocess_tests PRIVATE vsh_subprocess_component)
add_test(NAME vsh_subprocess_lifecycle COMMAND vsh_subprocess_tests $<TARGET_FILE:vsh_subprocess_probe>)
set_tests_properties(vsh_subprocess_lifecycle PROPERTIES TIMEOUT 40 RUN_SERIAL TRUE)
add_executable(vsh_subprocess_input_tests input_tests.cpp)
target_link_libraries(vsh_subprocess_input_tests PRIVATE vsh_subprocess_component)
add_test(NAME vsh_subprocess_input COMMAND vsh_subprocess_input_tests $<TARGET_FILE:vsh_subprocess_probe>)
set_tests_properties(vsh_subprocess_input PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
add_executable(vsh_subprocess_frame_mailbox_tests frame_mailbox_tests.cpp)
target_link_libraries(vsh_subprocess_frame_mailbox_tests PRIVATE vsh_subprocess_component)
add_test(NAME vsh_subprocess_frame_mailbox COMMAND vsh_subprocess_frame_mailbox_tests)
set_tests_properties(vsh_subprocess_frame_mailbox PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
find_package(Vulkan REQUIRED)
add_executable(vsh_subprocess_window_probe window_probe.cpp)
target_link_libraries(vsh_subprocess_window_probe PRIVATE vsh_subprocess_component Vulkan::Vulkan)
add_executable(vsh_subprocess_window_tests window_tests.cpp)
target_link_libraries(vsh_subprocess_window_tests PRIVATE vsh_subprocess_component)
add_test(NAME vsh_subprocess_embedded_vulkan COMMAND vsh_subprocess_window_tests $<TARGET_FILE:vsh_subprocess_window_probe>)
set_tests_properties(vsh_subprocess_embedded_vulkan PROPERTIES TIMEOUT 40 RUN_SERIAL TRUE)
# Parent integration: these reuse deterministic existing production-CPP fixtures;
# no physical device initialization. Standalone transport stays dependency-light.

  endforeach()
